      function renderFrameworkGrid() {
        const container = document.getElementById("frameworks-container");
        container.innerHTML = "";

        FRAMEWORKS.forEach(fw => {
          const isSelected = fw.id === selectedFramework;
          const card = document.createElement("div");
          card.className = `framework-card ${isSelected ? "selected" : ""}`;
          card.id = `card-${fw.id}`;
          card.onclick = () => selectFramework(fw.id);

          card.innerHTML = `
            
    
              
    
                
    ${fw.icon}
    
                
    ${isSelected ? "✓" : ""}
    
              
    
              
    ${fw.name}
    
              
    ${fw.tagline}

                ${fw.tags.map(t => `${t}`).join("")}
              
    
            
    
            
    
              Port: ${fw.defaultPort}
              Zero Config
            
    
          `;
          container.appendChild(card);

      }

      function selectFramework(id) {
        selectedFramework = id;
        document.querySelectorAll(".framework-card").forEach(c => {
          c.classList.remove("selected");
          const chk = c.querySelector(".framework-check");
          if (chk) chk.textContent = "";
        });
        const selectedCard = document.getElementById(`card-${id}`);
        if (selectedCard) {
          selectedCard.classList.add("selected");
          const chk = selectedCard.querySelector(".framework-check");
          if (chk) chk.textContent = "✓";
        }
        showToast(`Selected framework: ${id.toUpperCase()}`);
        updateExport();
      }

      function updateExport() {
        const projectName = document.getElementById("input-project-name").value.trim() || "polyglot-app";
        const appTitle = document.getElementById("input-app-title").value.trim() || "Polyglot App";
        const appDesc = document.getElementById("input-app-desc").value.trim() || "";
        const theme = document.getElementById("select-project-theme").value;
        const features = [];
        if (document.getElementById("feat-tailwind").checked) features.push("tailwind");
        if (document.getElementById("feat-typescript").checked) features.push("typescript");
        if (document.getElementById("feat-pwa").checked) features.push("pwa");
        if (document.getElementById("feat-mcp").checked) features.push("mcp");
        if (document.getElementById("feat-ci").checked) features.push("ci");

        currentGeneratedFiles = generateFullFrameworkProject(selectedFramework, projectName, appTitle, appDesc, theme, features);
        renderFileTree();
      }

      function renderFileTree() {
        const container = document.getElementById("file-tree-container");
        container.innerHTML = "";
        const filePaths = Object.keys(currentGeneratedFiles).sort();

        document.getElementById("file-count").textContent = filePaths.length;
        let totalBytes = 0;
        filePaths.forEach(p => totalBytes += (currentGeneratedFiles[p] || "").length);
        document.getElementById("bundle-size-tag").textContent = (totalBytes / 1024).toFixed(1) + " KB";

        filePaths.forEach((path, idx) => {
          const li = document.createElement("li");
          li.className = `file-tree-item ${path === activeFilePath || (!activeFilePath && idx === 0) ? "active" : ""}`;
          li.onclick = () => selectFile(path);

          let icon = "◆";
          if (path.endsWith(".json")) icon = "✦";
          else if (path.endsWith(".ts") || path.endsWith(".tsx") || path.endsWith(".js") || path.endsWith(".mjs")) icon = "▣";
          else if (path.endsWith(".astro")) icon = "★";
          else if (path.endsWith(".svelte")) icon = "⚡";
          else if (path.endsWith(".vue")) icon = "✦";
          else if (path.endsWith(".md")) icon = "▣";
          else if (path.endsWith(".rs")) icon = "✦";

          li.innerHTML = `${icon} ${path}`;
          container.appendChild(li);
        });

        if (!activeFilePath || !currentGeneratedFiles[activeFilePath]) {
          activeFilePath = filePaths[0] || "";
        }
        displayActiveFile();
      }

      function selectFile(path) {
        activeFilePath = path;
        document.querySelectorAll(".file-tree-item").forEach(el => {
          if (el.textContent.includes(path)) el.classList.add("active");
          else el.classList.remove("active");
        });
        displayActiveFile();
      }

      function displayActiveFile() {
        const codeDisplay = document.getElementById("code-display");
        const nameDisplay = document.getElementById("active-file-name");

        if (!activeFilePath || !currentGeneratedFiles[activeFilePath]) {
          nameDisplay.textContent = "No file selected";
          codeDisplay.textContent = "// Select a file to inspect";
          return;
        }

        nameDisplay.textContent = activeFilePath;
        codeDisplay.textContent = currentGeneratedFiles[activeFilePath];
        codeDisplay.classList.remove('locked');
      }

      function triggerZipDownload() {
        const projectName = document.getElementById("input-project-name").value.trim() || "polyglot-app";
        const fileList = Object.keys(currentGeneratedFiles).map(name => ({
          name,
          content: currentGeneratedFiles[name]
        }));

        const zipBlob = createZipArchive(fileList);
        const url = URL.createObjectURL(zipBlob);
        const a = document.createElement("a");
        a.href = url;
        a.download = `${projectName}-${selectedFramework}-bundle.zip`;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
        showToast(`▣ Downloaded ${projectName}-${selectedFramework}-bundle.zip!`);
      }

      document.getElementById("btn-download-zip").addEventListener("click", triggerZipDownload);
      document.getElementById("btn-copy-file").addEventListener("click", () => {
        if (activeFilePath && currentGeneratedFiles[activeFilePath]) {
          navigator.clipboard.writeText(currentGeneratedFiles[activeFilePath]).then(() => {
            showToast(`Copied ${activeFilePath} to clipboard!`);
          });
        }
      });

      function runTranspiler() {
        const inputCode = document.getElementById("transpile-input").value;
        const targetFw = document.getElementById("transpile-target").value;
        const outputEl = document.getElementById("transpile-output");

        let converted = inputCode;
        if (targetFw === "astro") {
          converted = `---\n// Astro 5 Component automatically transpiled\nconst { title = "Sovereign App", subtitle } = Astro.props;\n---\n\n` + inputCode.replace(/className=/g, "class=").replace(/onClick=\{[^}]+\}/g, 'onclick="alert(\'Action\')"');
        } else if (targetFw === "svelte") {
          converted = '<' + 'script lang="ts">\n  let { title = "", subtitle = "" } = $props();\n  let count = $state(0);\n<' + '/script>\n\n' + inputCode.replace(/onClick=\{[^}]+\}/g, 'on:click={() => count++}').replace(/className=/g, "class=");
        } else if (targetFw === "vue") {
          converted = '<' + 'script setup lang="ts">\nimport { ref } from "vue";\nconst count = ref(0);\n<' + '/script>\n\n';
        } else if (targetFw === "solid") {
          converted = inputCode.replace(/useState\(([^)]*)\)/g, 'createSignal($1)');
        } else if (targetFw === "htmx") {
          converted = `
    \n  \n
    `;
        }
        outputEl.textContent = converted;
      }
